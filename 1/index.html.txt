<script>
function colorClass(percent) {
  if (percent >= 80) return 'green';
  if (percent >= 60) return 'yellow';
  if (percent >= 30) return 'orange';
  return 'red';
}

function machColor(m) {
  if (m >= 3) return 'green';
  if (m === 2) return 'yellow';
  if (m === 1) return 'orange';
  return 'red';
}

function updateMask() {
  const stab = Number(document.getElementById('in-stab').value || 0);
  const mach = Number(document.getElementById('in-mach').value || 0);
  const fin  = Number(document.getElementById('in-final').value || 0);
  const harm = Number(document.getElementById('in-harm').value || 0);

  const delta = (harm + mach) / 2;

  // Stabilität
  document.getElementById('val-stab').textContent = stab + ' %';
  document.getElementById('stat-stab').textContent = stab >= 60 ? 'stabil' : 'instabil';
  document.getElementById('row-stab').className = 'row ' + colorClass(stab);

  // Machbarkeit
  document.getElementById('val-mach').textContent = '+' + mach;
  document.getElementById('stat-mach').textContent =
    mach === 3 ? 'voll machbar' :
    mach === 2 ? 'machbar mit Aufwand' :
    mach === 1 ? 'kritisch' : 'nicht machbar';
  document.getElementById('row-mach').className = 'row ' + machColor(mach);

  // Final
  document.getElementById('val-final').textContent = fin + ' %';
  document.getElementById('stat-final').textContent = fin >= 90 ? 'fast abgeschlossen' :
                                                      fin >= 60 ? 'in Arbeit' : 'frühe Phase';
  document.getElementById('row-final').className = 'row ' + colorClass(fin);

  // Harmonie
  document.getElementById('val-harm').textContent = harm + ' %';
  document.getElementById('stat-harm').textContent = harm >= 80 ? 'harmonisch' :
                                                     harm >= 60 ? 'ok' :
                                                     harm >= 30 ? 'angespannt' : 'disharmonisch';
  document.getElementById('row-harm').className = 'row ' + colorClass(harm);

  // Rettung (?)
  document.getElementById('val-rett').textContent = '? = ' + delta.toFixed(1);
  let rettText, rettColor;
  if (delta > 30) { rettText = 'rettbar'; rettColor = 'green'; }
  else if (delta >= 20) { rettText = 'möglich'; rettColor = 'yellow'; }
  else if (delta >= 10) { rettText = 'kritisch'; rettColor = 'orange'; }
  else { rettText = 'stoppen'; rettColor = 'red'; }
  document.getElementById('stat-rett').textContent = rettText;
  document.getElementById('row-rett').className = 'row ' + rettColor;

  // Gesamtstatus
  const allGreen = colorClass(stab)==='green' && machColor(mach)==='green' &&
                   colorClass(fin)==='green' && colorClass(harm)==='green' &&
                   rettColor==='green';
  const gesamtText = allGreen
    ? 'Projekt stabil, machbar und rettbar.'
    : 'Projekt benötigt Aufmerksamkeit.';
  document.getElementById('val-gesamt').textContent = '';
  document.getElementById('stat-gesamt').textContent = gesamtText;
  document.getElementById('row-gesamt').className = 'row ' + (allGreen ? 'green' : rettColor);
}

// initial
updateMask();
</script>
