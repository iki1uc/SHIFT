/* ================================ SOLL-PROFIL (CSS/JS) – FINAL ================================ */ // IST-Werte (Basis) const CSS_IST = { harmonie: 1, klarheit: 1, stabilitaet: 1 }; const JS_IST = { funktion: 1, harmonie: 1, performance: 1 }; // NEU-Werte (index.html = SOLL) const CSS_NEU = { harmonie: 2, klarheit: 2, stabilitaet: 2 }; const JS_NEU = { funktion: 2, harmonie: 2, performance: 2 }; // Technikprüfung – sagt IMMER, ob NEU > IST function technikBietetMehr(neu, alt) { if (neu > alt) { console.log("SOLL: Neue Technik bietet mehr als IST."); return true; } console.log("IST bleibt: Neue Technik bietet NICHT mehr."); return false; } // CSS-Prüfung function pruefeCSS(cssNeu, cssAlt) { const neuScore = cssNeu.harmonie + cssNeu.klarheit + cssNeu.stabilitaet; const altScore = cssAlt.harmonie + cssAlt.klarheit + cssAlt.stabilitaet; return technikBietetMehr(neuScore, altScore); } // JS-Prüfung function pruefeJS(jsNeu, jsAlt) { const neuScore = jsNeu.funktion + jsNeu.harmonie + jsNeu.performance; const altScore = jsAlt.funktion + jsAlt.harmonie + jsAlt.performance; return technikBietetMehr(neuScore, altScore); } // CUBE_3 entscheidet über SOLL function entscheideSOLL(cssNeu, cssAlt, jsNeu, jsAlt) { const cssOK = pruefeCSS(cssNeu, cssAlt); const jsOK = pruefeJS(jsNeu, jsAlt); if (cssOK && jsOK) { console.log("CUBE_3: SOLL aktiviert (NEU > IST)."); document.getElementById("cube-status").innerText += " | SOLL aktiv."; return "SOLL"; } console.log("CUBE_3: IST bleibt aktiv (NEU ≤ IST)."); document.getElementById("cube-status").innerText += " | IST bleibt."; return "IST"; } // Automatische Prüfung beim Aktivieren von CUBE_3 function pruefeSOLLBeiCube3() { if (window.currentCube === 3) { entscheideSOLL(CSS_NEU, CSS_IST, JS_NEU, JS_IST); } }
